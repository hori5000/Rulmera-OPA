# 서버 환경 점검

```bash
echo "===== GPU ====="
nvidia-smi --query-gpu=name,memory.total,driver_version,compute_cap --format=csv

echo
echo "===== OS ====="
cat /etc/os-release

echo
echo "===== PYTHON ====="
python3 --version

echo
echo "===== DOCKER ====="
docker --version 2>&1
```
